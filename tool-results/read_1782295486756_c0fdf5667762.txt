    88→// ─── Helper: load script ───────────────────────────────────────────────
    89→function loadScript(src: string): Promise<void> {
    90→  return new Promise((resolve, reject) => {
    91→    // Check if already loaded
    92→    const existing = document.querySelector(`script[src="${src}"]`);
    93→    if (existing) {
    94→      resolve();
    95→      return;
    96→    }
    97→    const s = document.createElement('script');
    98→    s.src = src;
    99→    s.crossOrigin = 'anonymous';
   100→    s.onload = () => resolve();
   101→    s.onerror = () => reject(new Error(`Failed to load: ${src}`));
   102→    document.head.appendChild(s);
   103→  });
   104→}

   106→// ─── Helper: format time MM:SS ────────────────────────────────────────
   107→function formatTime(seconds: number): string {
   108→  const m = Math.floor(seconds / 60);
   109→  const s = seconds % 60;
   110→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
   111→}

   488→  // ─── Handle pose results ────────────────────────────────────────────
   489→  const handlePoseResults = useCallback(
   490→    (results: unknown) => {

   502→

   504→      if (!video) return;
   505→
   506→      // Match canvas to displayed video size
   507→      canvas.width = video.videoWidth;
   508→      canvas.height = video.videoHeight;
   509→      ctx.clearRect(0, 0, canvas.width, canvas.height);
   510→
   511→      // Calculate angles
   512→      const angles = calculateAllAngles(landmarks);
   513→
   514→      // Update store angles
   515→      Object.entries(angles).forEach(([joint, angle]) => {
   516→        store.setLiveAngle(joint, angle);
   517→      });
   518→
   519→      // Store landmarks
   520→      store.setSkeletonLandmarks(landmarks.map((l) => ({ x: l.x, y: l.y, z: l.z })));
   521→
   522→      // Determine colors for connections
   523→      const targetJoints = selectedExercise?.targetJoints || [];
   524→
   525→      // Build a map of joint index -> color based on target joints
   526→      const jointColorMap: Record<number, string> = {};
   527→      const JOINT_MAP: Record<string, number> = {
   528→        left_shoulder: LANDMARKS.LEFT_SHOULDER,
   529→        right_shoulder: LANDMARKS.RIGHT_SHOULDER,
   530→        left_elbow: LANDMARKS.LEFT_ELBOW,
   531→        right_elbow: LANDMARKS.RIGHT_ELBOW,
   532→        left_wrist: LANDMARKS.LEFT_WRIST,
   533→        right_wrist: LANDMARKS.RIGHT_WRIST,
   534→        left_hip: LANDMARKS.LEFT_HIP,
   535→        right_hip: LANDMARKS.RIGHT_HIP,
   536→        left_knee: LANDMARKS.LEFT_KNEE,
   537→        right_knee: LANDMARKS.RIGHT_KNEE,
   538→        left_ankle: LANDMARKS.LEFT_ANKLE,
   539→        right_ankle: LANDMARKS.RIGHT_ANKLE,
   540→      };
   541→
   542→      // Color for each target joint
   543→      const jointStatusColor: Record<string, string> = {};
   544→      targetJoints.forEach((tj) => {
   545→        const currentAngle = angles[tj.name];
   546→        if (currentAngle !== undefined) {
   547→          if (currentAngle >= tj.minAngle && currentAngle <= tj.maxAngle) {
   548→            jointStatusColor[tj.name] = '#10b981'; // green
   549→          } else {
   550→            const dev = Math.min(
   551→              Math.abs(currentAngle - tj.minAngle),
   552→              Math.abs(currentAngle - tj.maxAngle)
   553→            );
   554→            jointStatusColor[tj.name] = dev <= 15 ? '#f59e0b' : '#ef4444'; // amber / red
   555→          }
   556→        }
   557→      });
   558→
   559→      // Map joint name to color
   560→      Object.entries(jointStatusColor).forEach(([name, color]) => {
   561→        const idx = JOINT_MAP[name];
   562→        if (idx !== undefined) {
   563→          jointColorMap[idx] = color;
   564→        }
   565→      });
   566→
   567→      // Draw skeleton connections
   568→      SKELETON_CONNECTIONS.forEach(([i, j]) => {
   569→        const lm1 = landmarks[i];
   570→        const lm2 = landmarks[j];
   571→        if (!lm1 || !lm2) return;
   572→        if ((lm1.visibility ?? 0) < 0.5 || (lm2.visibility ?? 0) < 0.5) return;
